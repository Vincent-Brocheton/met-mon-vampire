> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal :** le conte rédige le wiki (une page par entrée du référentiel, des pages libres), avec brouillon à part et publication versionnée ; les joueurs le lisent (accueil, rubrique, page, recherche, hors ligne) et ne voient que les pages de leur clan, de leur secte ou ouvertes à tous.

**Architecture :**
- **Données :** `wiki/{id}` (version publiée), `wiki/{id}/versions/{n}` (copie de chaque publication), `wikiDrafts/{id}` (brouillon, équipe seulement), `users/{uid}.wikiAudiences` (accès par clan et par secte).

- **Dépôt `WikiRepository`** et providers ; synchronisation des audiences (`wikiAudienceSyncProvider`) lancée par la liste des fiches et le wiki du conte.
- **Écrans :** lecture partagée joueur / conte (`WikiNav` donne les adresses, `?apercu=1` pour l'aperçu joueur), liste du conte, édition.

**Tech Stack :** inchangée (aucune dépendance nouvelle).

```dart
import 'package:flutter/services.dart';

/// Bloc du texte d'une page.
sealed class WikiBlock {
  const WikiBlock();
}

/// Titre `## …`.
class WikiHeading extends WikiBlock {
  const WikiHeading(this.text);
  final String text;
}

class WikiPara extends WikiBlock {
  const WikiPara(this.text);
  final String text;
}

/// Encadré « § Règle ».
class WikiRuleBox extends WikiBlock {
  const WikiRuleBox(this.text);
  final String text;
}

class WikiList extends WikiBlock {
  const WikiList(this.items);
  final List<String> items;
}

/// Tableau ; la première ligne sert d'en-tête.
class WikiTable extends WikiBlock {
  const WikiTable(this.rows);
  final List<List<String>> rows;
}

final _separator = RegExp(r'^:?-{3,}:?$');

/// Découpe le Markdown léger du wiki : `## titre`, `- liste`, `§ encadré`, `| a | b |`,
/// paragraphes séparés par une ligne vide.
List<WikiBlock> parseWiki(String source) {
  final blocks = <WikiBlock>[];
  final para = <String>[];
  List<String>? list;
  List<List<String>>? table;
  void flush() {
    if (para.isNotEmpty) blocks.add(WikiPara(para.join(' ')));
    para.clear();
    if (list case final l?) blocks.add(WikiList(l));
    if (table case final t?) blocks.add(WikiTable(t));
    list = null;
    table = null;
  }

  for (final raw in source.split('\n')) {
    final line = raw.trim();
    if (line.isEmpty) {
      flush();
    } else if (line.startsWith('## ')) {
      flush();
      blocks.add(WikiHeading(line.substring(3).trim()));
    } else if (line.startsWith('§')) {
      flush();
      blocks.add(WikiRuleBox(line.substring(1).trim()));
    } else if (line.startsWith('- ')) {
      if (list == null) flush();
      (list ??= []).add(line.substring(2).trim());
    } else if (line.length > 1 && line.startsWith('|') && line.endsWith('|')) {
      if (table == null) flush();
      final rows = table ??= [];
      final cells = [for (final c in line.substring(1, line.length - 1).split('|')) c.trim()];
      if (!cells.every(_separator.hasMatch)) rows.add(cells);
    } else {
      if (list != null || table != null) flush();
      para.add(line);
    }
  }
  flush();
  return blocks;
}

/// Morceau de texte en ligne : `**gras**`, `*italique*`, `[[lien]]`.
class WikiSpan {
  const WikiSpan(this.text, {this.bold = false, this.italic = false, this.link = false});
  final String text;
  final bool bold;
  final bool italic;
  final bool link;
}

final _inline = RegExp(r'\*\*(.+?)\*\*|\*(.+?)\*|\[\[(.+?)\]\]');

/// Une syntaxe non fermée reste du texte.
List<WikiSpan> parseInline(String s) {
  final out = <WikiSpan>[];
  var at = 0;
  for (final m in _inline.allMatches(s)) {
    if (m.start > at) out.add(WikiSpan(s.substring(at, m.start)));
    out.add(m[1] != null
        ? WikiSpan(m[1]!, bold: true)
        : m[2] != null
            ? WikiSpan(m[2]!, italic: true)
            : WikiSpan(m[3]!.trim(), link: true));
    at = m.end;
  }
  if (at < s.length) out.add(WikiSpan(s.substring(at)));
  return out;
}

String _plain(String s) => parseInline(s).map((x) => x.text).join();

/// Texte sans balises (recherche, extraits).
String plainText(String source) => [
      for (final b in parseWiki(source))
        switch (b) {
          WikiHeading(:final text) || WikiPara(:final text) || WikiRuleBox(:final text) => _plain(text),
          WikiList(:final items) => items.map(_plain).join(' '),
          WikiTable(:final rows) => [for (final r in rows) r.map(_plain).join(' ')].join(' '),
        },
    ].join('\n');

/// Titres `##` de la page, pour le sommaire.
List<String> headingsOf(String source) => [for (final b in parseWiki(source)) if (b is WikiHeading) b.text];

/// Entoure la sélection de [open] et [close] ; sans sélection, insère [placeholder] entouré, et le sélectionne.
TextEditingValue wrapSelection(TextEditingValue v, String open, String close, String placeholder) {
  final sel = v.selection.isValid ? v.selection : TextSelection.collapsed(offset: v.text.length);
  final inner = sel.isCollapsed ? placeholder : sel.textInside(v.text);
  final start = sel.start + open.length;
  return TextEditingValue(
    text: sel.textBefore(v.text) + open + inner + close + sel.textAfter(v.text),
    selection: TextSelection(baseOffset: start, extentOffset: start + inner.length),
  );
}

/// Ajoute [prefix] au début de la ligne du curseur (titre, liste, encadré).
TextEditingValue prefixLine(TextEditingValue v, String prefix) {
  final at = v.selection.isValid ? v.selection.start : v.text.length;
  final lineStart = at == 0 ? 0 : v.text.lastIndexOf('\n', at - 1) + 1;
  return TextEditingValue(
    text: v.text.substring(0, lineStart) + prefix + v.text.substring(lineStart),
    selection: TextSelection.collapsed(offset: at + prefix.length),
  );
}

/// Insère [block] sur ses propres lignes, après la ligne du curseur.
TextEditingValue insertBlock(TextEditingValue v, String block) {
  final at = v.selection.isValid ? v.selection.end : v.text.length;
  final lineEnd = v.text.indexOf('\n', at);
  final cut = lineEnd < 0 ? v.text.length : lineEnd;
  final before = v.text.substring(0, cut);
  final lead = before.isEmpty ? '' : (before.endsWith('\n') ? '\n' : '\n\n');
  return TextEditingValue(
    text: '$before$lead$block\n${v.text.substring(cut)}',
    selection: TextSelection.collapsed(offset: before.length + lead.length + block.length),
  );
}
```

```dart
/// Éléments lisibles, triés par titre. Joueur : entrées proposées et pages reçues (déjà filtrées par les règles) ;
/// équipe : toutes les entrées, les pages orphelines et les brouillons.
List<WikiItem> wikiIndex(Rulebook rb, List<WikiPage> pages, {required bool staff, List<WikiPage> drafts = const []}) {
  final published = {for (final p in pages) p.id: p};
  final drafted = {for (final d in drafts) d.id: d};
  final items = <WikiItem>[];
  final seen = <String>{};
  for (final r in rubrics) {
    for (final cat in r.cats) {
      for (final e in rb.all(cat)) {
        if (!staff && !e.state.offered) continue;
        final id = rulePageId(cat, e);
        if (!seen.add(id)) continue;
        items.add(WikiItem(id: id, title: e.name, rubric: r.id, cat: cat, entry: e, page: published[id], draft: drafted[id]));
      }
    }
  }
  for (final p in pages) {
    if (seen.contains(p.id) || (p.cat != null && !staff)) continue;
    seen.add(p.id);
    items.add(WikiItem(
      id: p.id,
      title: p.title,
      rubric: p.cat == null ? rubricById(p.rubric).id : rubricOfCat(p.cat!),
      cat: p.cat,
      page: p,
      draft: drafted[p.id],
      orphan: p.cat != null,
    ));
  }
  for (final d in drafts) {
    if (!seen.add(d.id)) continue;
    items.add(WikiItem(
      id: d.id,
      title: d.title,
      rubric: d.cat == null ? rubricById(d.rubric).id : rubricOfCat(d.cat!),
      cat: d.cat,
      draft: d,
      orphan: d.cat != null,
    ));
  }
  return items..sort((a, b) => fold(a.title).compareTo(fold(b.title)));
}

String _excerpt(String text, int at, int length) {
  final start = max(0, at - 40);
  final end = min(text.length, at + length + 60);
  final s = text.substring(start, end).replaceAll(RegExp(r'\s+'), ' ').trim();
  return '${start > 0 ? '… ' : ''}$s${end < text.length ? ' …' : ''}';
}

/// Sans casse ni accents ; les titres (et noms VO) d'abord, puis le texte.
List<SearchHit> search(List<WikiItem> items, String query) {
  final q = fold(query.trim());
  if (q.isEmpty) return const [];
  final byTitle = <SearchHit>[];
  final byText = <SearchHit>[];
  for (final i in items) {
    final text = searchText(i);
    final at = fold(text).indexOf(q);
    final excerpt = at < 0 ? _excerpt(text, 0, 0) : _excerpt(text, at, q.length);
    if (fold(i.title).contains(q) || fold(i.vo ?? '').contains(q)) {
      byTitle.add(SearchHit(i, excerpt));
    } else if (at >= 0) {
      byText.add(SearchHit(i, excerpt));
    }
  }
  return [...byTitle, ...byText];
}

/// Page visible de ce titre : un lien vers une page cachée reste du texte.
WikiItem? resolveLink(String title, List<WikiItem> items) => items.where((i) => nameKey(i.title) == nameKey(title)).firstOrNull;

String? _fieldText(RuleField f, Object? v) {
  if (f.type == FieldType.flag) return v == true ? 'Oui' : null;
  if (f.type == FieldType.rows) {
    final rows = [
      for (final r in (v as List?) ?? const [])
        if (r is Map)
          [
            for (final rf in f.rowFields)

          ].join(' · '),

    return rows.isEmpty ? null : rows.join(' ; ');
  }
  final s = displayValue(f, v);
  return s == '—' ? null : s;
}
```

```dart
/// `wiki/{id}` (publiée), `wiki/{id}/versions/{n}` (copies), `wikiDrafts/{id}` (brouillons), `users/{uid}.wikiAudiences`.
class WikiRepository {
  WikiRepository(this._db);

  final FirebaseFirestore _db;

  CollectionReference<Map<String, dynamic>> get _pages => _db.collection('wiki');
  CollectionReference<Map<String, dynamic>> get _drafts => _db.collection('wikiDrafts');

  static List<WikiPage> _list(QuerySnapshot<Map<String, dynamic>> q) => [for (final d in q.docs) WikiPage.fromMap(d.id, d.data())];

  /// Toutes les pages publiées (équipe).
  Stream<List<WikiPage>> watchAll() => _pages.snapshots().map(_list);

  /// Pages d'un joueur : le filtre que testent les règles.
  Stream<List<WikiPage>> watchVisible(List<String> audiences) =>
      _pages.where('audience', whereIn: [audienceAll, ...audiences]).snapshots().map(_list);

  Stream<List<WikiPage>> watchDrafts() => _drafts.snapshots().map(_list);

  String newId() => _pages.doc().id;

  Map<String, dynamic> _by(Actor by) => {'updatedAt': FieldValue.serverTimestamp(), 'updatedByUid': by.uid, 'updatedByName': by.name};

  /// [page].version : version publiée sur laquelle part le brouillon.
  Future<void> saveDraft(WikiPage page, Actor by) =>
      _drafts.doc(page.id).set({...page.contentMap(), 'baseVersion': page.version, ..._by(by)});

  Future<void> discardDraft(String id) => _drafts.doc(id).delete();

  /// Publie [page] en version [page].version + 1, en un lot : page, copie de version, brouillon supprimé.
  /// Refusé si un autre conteur a publié entre-temps.
  Future<void> publish(WikiPage page, Actor by, {String note = ''}) {
    final n = page.version + 1;
    final batch = _db.batch()
      ..set(_pages.doc(page.id), {...page.contentMap(), 'version': n, ..._by(by)})
      ..set(_pages.doc(page.id).collection('versions').doc('$n'),
          {...page.contentMap(), 'note': note.trim(), 'byUid': by.uid, 'byName': by.name, 'at': FieldValue.serverTimestamp()})
      ..delete(_drafts.doc(page.id));
    return batch.commit();
  }

  /// Supprime la page publiée et son brouillon ; les versions restent.
  Future<void> delete(String id) {
    final batch = _db.batch()
      ..delete(_pages.doc(id))
      ..delete(_drafts.doc(id));
    return batch.commit();
  }

  Future<void> setAudiences(Map<String, List<String>> changes) {
    final batch = _db.batch();
    for (final MapEntry(key: uid, value: list) in changes.entries) {
      batch.update(_db.collection('users').doc(uid), {'wikiAudiences': list});
    }
    return batch.commit();

@riverpod
Future<void> wikiAudienceSync(Ref ref) async {
  final manages = ref.watch(currentUserProvider.select((u) => u.value?.role.managesAccounts ?? false));
  if (!manages) return;
  final users = await ref.watch(allUsersProvider.future);
  final sheets = await ref.watch(allCharactersProvider.future);
  final changes = audienceChanges(users, sheets);
  if (changes.isEmpty) return;
  try {
    await ref.read(wikiRepositoryProvider).setAudiences(changes);
  } catch (e) {
    // Retentée à la prochaine ouverture d'un écran du conte.
    debugPrint('Accès du wiki non synchronisés : $e');
  }
```

```dart
  InlineSpan _span(WikiSpan s, TextStyle? style, {required bool links}) {
    final id = s.link && links ? target(s.text) : null;
    if (id == null) {
      return TextSpan(text: s.text, style: TextStyle(fontWeight: s.bold ? FontWeight.w700 : null, fontStyle: s.italic ? FontStyle.italic : null));

    );

        border: TableBorder.all(color: AppColors.border),

            TableRow(children: [
              for (var c = 0; c < width; c++)
                Padding(padding: const EdgeInsets.all(8), child: _rich(c < r.length ? r[c] : '', n == 0 ? head : body, links: false)),
            ]),
        ],
      ),
    );
  }

    return Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
```

```dart
/// Édition d'une page (C-WikiEdit) : texte et aperçu, visibilité ; brouillon à part, publication en un lot.
class WikiEditScreen extends ConsumerStatefulWidget {
  const WikiEditScreen({super.key, required this.pageId});
  final String pageId;

  @override
  ConsumerState<WikiEditScreen> createState() => _WikiEditScreenState();
}

class _WikiEditScreenState extends ConsumerState<WikiEditScreen> {
  final _title = TextEditingController();
  final _body = TextEditingController();
  final _source = TextEditingController();
  var _audience = audienceAll;
  var _rubric = 'systeme';
  var _loaded = false;
  var _busy = false;
  var _showPreview = false;
  late final String _id = widget.pageId == newPageId ? ref.read(wikiRepositoryProvider).newId() : widget.pageId;

  @override
  void dispose() {
    _title.dispose();
    _body.dispose();
    _source.dispose();
    super.dispose();
  }

  void _load(WikiPage? from, WikiItem? item) {
    _title.text = from?.title ?? item?.title ?? '';
    _body.text = from?.body ?? '';
    _source.text = from?.source ?? item?.entry?.source ?? '';
    _audience = from?.audience ?? audienceAll;
    _rubric = item?.rubric ?? 'systeme';
    _loaded = true;
  }

  WikiPage _current(WikiItem? item) {
    final rule = item?.isRule ?? false;
    return WikiPage(
      id: _id,
      title: rule ? item!.title : _title.text.trim(),
      rubric: rule ? item!.rubric : _rubric,
      audience: _audience,
      body: _body.text,
      source: _source.text.trim(),
      cat: item?.cat,
      entryId: item?.entry != null ? entryKey(item!.entry!) : item?.page?.entryId ?? item?.draft?.entryId,
      version: item?.draft?.version ?? item?.page?.version ?? 0,
    );
  }

  bool _valid(WikiItem? item) {
    if ((item?.isRule ?? false) || _title.text.trim().isNotEmpty) return true;
    ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Titre obligatoire.')));
    return false;
  }

  Actor get _by => actorOf(ref.read(currentUserProvider).value)!;

  /// Vrai si l'écriture a réussi ; sinon le formulaire reste tel quel.
  Future<bool> _run(Future<void> Function() write, String done, {String refused = 'Enregistrement refusé : réessayez.'}) async {
    setState(() => _busy = true);
    try {
      await write();
      if (mounted) ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(done)));
      return true;
    } catch (e) {
      final denied = e is FirebaseException && e.code == 'permission-denied';
      if (mounted) ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(denied ? refused : 'Enregistrement refusé : réessayez.')));
      return false;
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  /// Une nouvelle page prend son adresse définitive une fois enregistrée.
  void _openSaved() {
    if (mounted && widget.pageId == newPageId) context.go('/conteur/wiki/edition/$_id');
  }

  Future<void> _saveDraft(WikiItem? item) async {
    if (!_valid(item)) return;
    if (await _run(() => ref.read(wikiRepositoryProvider).saveDraft(_current(item), _by), 'Brouillon enregistré.')) _openSaved();
  }

  Future<void> _publish(WikiItem? item) async {
    if (!_valid(item)) return;
    final note = await showDialog<String>(context: context, builder: (_) => const _NoteDialog());
    if (note == null || !mounted) return;
    if (await _run(() => ref.read(wikiRepositoryProvider).publish(_current(item), _by, note: note), 'Page publiée.',
        refused: 'Cette page a été publiée entre-temps. Rechargez-la avant de publier.')) {
      _openSaved();
    }
  }

  Future<void> _discard(WikiItem item) async {
    if (!await confirm(context, title: 'Abandonner le brouillon', body: 'Les modifications non publiées seront perdues.', action: 'Abandonner')) return;
    if (!await _run(() => ref.read(wikiRepositoryProvider).discardDraft(_id), 'Brouillon abandonné.') || !mounted) return;
    if (item.page == null) {
      context.go('/conteur/wiki');
    } else {
      setState(() => _load(item.page, item));
    }
  }

  Future<void> _delete() async {
    if (!await confirm(context,
        title: 'Supprimer la page', body: 'La page publiée et son brouillon seront supprimés. Les versions restent dans l’historique.', action: 'Supprimer')) {
      return;
    }
    if (await _run(() => ref.read(wikiRepositoryProvider).delete(_id), 'Page supprimée.') && mounted) context.go('/conteur/wiki');
  }

  void _apply(TextEditingValue Function(TextEditingValue) edit) => setState(() => _body.value = edit(_body.value));

  @override
  Widget build(BuildContext context) {
    final manages = ref.watch(currentUserProvider.select((u) => u.value?.role.managesAccounts ?? false));
    final rb = ref.watch(rulebookProvider);
    final drafts = ref.watch(wikiDraftsProvider);
    if (rb == null) return const Center(child: CircularProgressIndicator());
    return asyncView(
      ref.watch(wikiPagesProvider),
      (pages) => asyncView(drafts, (drafts) {
        final items = wikiIndex(rb, pages, drafts: drafts, staff: true);
        final item = items.where((i) => i.id == _id).firstOrNull;
        if (item == null && widget.pageId != newPageId) {
          return const EmptyState(kind: EmptyKind.notFound, title: 'Page introuvable', message: 'Cette page n’existe pas.');
        }
        if (!_loaded) _load(item?.draft ?? item?.page, item);
        return _form(context, rb, items, item, manages: manages);
      }, onRetry: () => ref.invalidate(wikiDraftsProvider)),
      onRetry: () => ref.invalidate(wikiPagesProvider),
    );
  }
```
