> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal :** compléter le wiki : historique des versions d'une page (comparaison, restauration), chronologie publique (événements « Public » de toutes les fiches) et personnalisation de la lecture pour le joueur (« Pour {nom} », « Vous l’avez », prochain niveau, « Préparer l’achat »).

**Architecture :**
- **Événements :** nouveau champ `charName` écrit avec chaque événement ; lecture des événements publics de toutes les fiches par `collectionGroup('events')` (règle de groupe + index de champ).
- **Chronologie :** page virtuelle `chronologie` (rubrique « La chronique ») dans l'index du wiki, rendue par `ChronologyView`.
- **Historique :** `word_diff.dart` (calcul pur), `WikiVersion`, `watchVersions`, écran `WikiHistoryScreen`, restauration par une publication normale.
- **Personnalisation :** calculs purs dans `wiki_personal.dart` ; lecture joueur enrichie (accueil, rubrique, page) à partir de ses fiches de PJ actives ; rien côté conte ni en aperçu.

**Tech Stack :** inchangée.

**Spec :** `docs/superpowers/specs/2026-10-11-wiki-design.md` (sections 9b). Le lot 9a est fusionné (`main`, commit `ae9ba79`).

**Maquettes :** `C-WikiHistorique`, `J-Wiki` (bloc « Pour Isaure »), `J-WikiPage` (encadré, « Vous l’avez »), `J-WikiRubrique` (puce « En clan pour Isaure ») et leurs versions `-mobile`.

```dart
/// Années de la chronologie, de la plus récente à la plus ancienne, chacune avec ses événements dans l'ordre des fiches.
List<(int, List<StoryEvent>)> timelineGroups(List<StoryEvent> events) {
  final sorted = [...events]..sort(compareEvents);
```

```dart
import 'dart:math';

enum DiffKind { same, added, removed }

typedef DiffPart = ({DiffKind kind, String text});

final _token = RegExp(r'\s+|\S+');

/// Différences mot à mot entre deux textes (plus longue sous-suite commune), espaces et sauts de ligne compris.
/// Les morceaux voisins de même nature sont fusionnés.
// ponytail: tableau n × m ; au-delà de 4 millions de cases, tout l'ancien texte est retiré et le nouveau ajouté.
List<DiffPart> wordDiff(String before, String after) {
  final a = [for (final m in _token.allMatches(before)) m[0]!];
  final b = [for (final m in _token.allMatches(after)) m[0]!];
  final out = <DiffPart>[];
  void add(DiffKind kind, String text) {
    if (out.isNotEmpty && out.last.kind == kind) {
      out[out.length - 1] = (kind: kind, text: out.last.text + text);
    } else {
      out.add((kind: kind, text: text));
    }
  }

  if (a.length * b.length > 4000000) {
    if (before.isNotEmpty) add(DiffKind.removed, before);
    if (after.isNotEmpty) add(DiffKind.added, after);
    return out;
  }
  // lcs[i][j] : longueur de la plus longue sous-suite commune de a[i..] et b[j..].
  final lcs = List.generate(a.length + 1, (_) => List.filled(b.length + 1, 0));
  for (var i = a.length - 1; i >= 0; i--) {
    for (var j = b.length - 1; j >= 0; j--) {
      lcs[i][j] = a[i] == b[j] ? lcs[i + 1][j + 1] + 1 : max(lcs[i + 1][j], lcs[i][j + 1]);
    }
  }
  var i = 0;
  var j = 0;
  while (i < a.length && j < b.length) {
    if (a[i] == b[j]) {
      add(DiffKind.same, a[i]);
      i++;
      j++;
    } else if (lcs[i + 1][j] >= lcs[i][j + 1]) {
      add(DiffKind.removed, a[i++]);
    } else {
      add(DiffKind.added, b[j++]);
    }
  }
  while (i < a.length) {
    add(DiffKind.removed, a[i++]);
  }
  while (j < b.length) {
    add(DiffKind.added, b[j++]);
  }
  return out;
}
```

```dart
/// Fiches qui personnalisent la lecture du wiki : les PJ actifs du joueur.
List<Character> activeSheets(Iterable<Character> sheets) =>
    [for (final c in sheets) if (c.kind == CharacterKind.pj && c.status == CharacterStatus.active) c];

String dots(int n) => '●' * n;

int disciplineLevel(Character c, String name) => c.disciplines.where((d) => nameKey(d.name) == nameKey(name)).firstOrNull?.level ?? 0;

/// Page de règle de cette catégorie et de ce nom, si elle est lisible.
WikiItem? itemFor(String cat, String? name, List<WikiItem> items) =>
    name == null ? null : items.where((i) => i.cat == cat && nameKey(i.title) == nameKey(name)).firstOrNull;

typedef MineLine = ({String title, String note, WikiItem? item});

/// Lignes du bloc « Pour {nom} » de l'accueil.
List<MineLine> mineLines(Character c, List<WikiItem> items) => [
      for (final d in c.disciplines)
        (title: d.name, note: '${dots(d.level)} · ${d.level == 1 ? 'votre pouvoir' : 'vos ${d.level} pouvoirs'}', item: itemFor('disciplines', d.name, items)),
      if (c.clan case final clan? when clan.trim().isNotEmpty) (title: clan, note: 'votre clan', item: itemFor('clans', clan, items)),
      for (final m in c.merits) (title: m.name, note: 'votre atout', item: itemFor('merits', m.name, items)),
      for (final f in c.flaws) (title: f.name, note: 'votre handicap', item: itemFor('flaws', f.name, items)),
    ];

String _clanWord(Character c, String discipline, Rulebook rb) => inClan(c, discipline, rb: rb) ? 'en clan' : 'hors clan';

int _levelCost(Character c, String discipline, int level, Rulebook rb) =>
    costOf(c, XpItem(XpKind.discipline, discipline, level - 1, level, 0), rb: rb);

typedef NextStep = ({String heading, String? next});

/// Encadré d'une fiche sur la page d'une discipline : niveau, prochain pouvoir et son coût (rien au-delà de 5).
NextStep nextStep(Character c, RuleEntry discipline, Rulebook rb) {
  final level = disciplineLevel(c, discipline.name);
  final heading = level == 0 ? '${c.name} · ${discipline.name} : pas encore apprise' : '${c.name} · ${discipline.name} ${dots(level)}';
  if (level >= 5) return (heading: heading, next: null);
  final power = disciplinePowers(discipline).where((p) => p.level == level + 1 && !p.elder).firstOrNull;
  final cost = _levelCost(c, discipline.name, level + 1, rb);
  return (heading: heading, next: 'Prochain niveau : ${power == null ? '' : '${power.name}, '}$cost XP ${_clanWord(c, discipline.name, rb)}.');
}

/// La fiche a ce pouvoir : niveau atteint (les pouvoirs d'anciens ne se déduisent pas du niveau).
bool owns(Character c, Power p, String discipline) => !p.elder && disciplineLevel(c, discipline) >= p.level;

/// « 9 XP en clan » : coût de ce niveau pour la fiche.
String powerCost(Character c, Power p, String discipline, Rulebook rb) =>
    '${_levelCost(c, discipline, p.level, rb)} XP ${_clanWord(c, discipline, rb)}';

/// Mention d'une carte de discipline : « ●● · en clan » si la fiche la possède.
String? cardMention(Character c, WikiItem i, Rulebook rb) {
  if (i.cat != 'disciplines') return null;
  final level = disciplineLevel(c, i.title);
  return level == 0 ? null : '${dots(level)} · ${_clanWord(c, i.title, rb)}';
}

/// Puce « En clan pour {nom} ».
bool inClanFor(Character c, WikiItem i, Rulebook rb) => i.cat == 'disciplines' && inClan(c, i.title, rb: rb);
```

```dart
                Padding(padding: const EdgeInsets.all(8), child: _rich(c < r.length ? r[c] : '', n == 0 ? head : body, links: false)),
            ]),
        ],
      ),
    );
  }
```

```dart
      'mis à jour le ${formatDay(p.updatedAt)}${p.updatedByName == null ? '' : ' par ${p.updatedByName}'}',
    ].join(' · ');

/// Page du wiki (J-WikiPage, J-WikiLore) : données du référentiel, texte, sommaire ; pour une discipline,
/// un encadré par fiche active du joueur. Une page inexistante ou cachée affiche le même message.
class WikiPageScreen extends ConsumerStatefulWidget {
  const WikiPageScreen({super.key, required this.nav, required this.pageId});
  final WikiNav nav;
  final String pageId;

  @override
  ConsumerState<WikiPageScreen> createState() => _WikiPageScreenState();
}

class _WikiPageScreenState extends ConsumerState<WikiPageScreen> {
  final _keys = <GlobalKey>[];

  List<GlobalKey> _keysFor(int n) {
    while (_keys.length < n) {
      _keys.add(GlobalKey());
    }
    return _keys.sublist(0, n);
  }

  @override
  Widget build(BuildContext context) {
    final t = Theme.of(context).textTheme;
    final nav = widget.nav;
    final manages = ref.watch(currentUserProvider.select((u) => u.value?.role.managesAccounts ?? false));
    final mine = mineFor(ref, nav);
    return WikiLoader(
      nav: nav,
      builder: (context, items, rb) {
        final item = items.where((i) => i.id == widget.pageId).firstOrNull;
        if (item == null) {
          return const EmptyState(kind: EmptyKind.notFound, title: 'Page introuvable', message: 'Cette page n’existe pas ou ne vous est pas ouverte.');
        }
        final page = item.page;
        final body = page?.body ?? '';
        final heads = headingsOf(body);
        final keys = _keysFor(heads.length);
        final r = rubricById(item.rubric);
        final banner = page == null ? null : reservedBanner(page.audience, rb);
        final discipline = item.cat == 'disciplines' ? item.entry : null;
        final article = Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
          WikiCrumbs([('Wiki', nav.home), (r.label, nav.rubric(r.id))], item.title),
          const SizedBox(height: 8),
          Wrap(spacing: 14, crossAxisAlignment: WrapCrossAlignment.end, children: [
            Text(item.title, style: isWide(context) ? t.displaySmall : t.headlineMedium),
            if ((item.vo ?? '').isNotEmpty) Text(item.vo!, style: t.bodyMedium?.copyWith(color: AppColors.textMuted, fontStyle: FontStyle.italic)),
          ]),
          if (banner != null) ...[
            const SizedBox(height: 14),
            Container(
              key: const Key('wk-banner'),
              padding: const EdgeInsets.symmetric(horizontal: 14, vertical: 10),
              decoration: BoxDecoration(color: const Color(0xFF2A2240), borderRadius: BorderRadius.circular(6)),
              child: Text(banner, style: t.bodyMedium?.copyWith(color: AppColors.narrator)),
            ),
          ],
          if (discipline != null)
            for (final c in mine)
              if (nextStep(c, discipline, rb) case final step) ...[
                const SizedBox(height: 14),
                Panel(
                  key: Key('wk-next-${c.id}'),
                  padding: const EdgeInsets.all(18),
                  child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
                    Text(step.heading, style: t.titleSmall),
                    if (step.next case final next?) ...[
                      const SizedBox(height: 4),
                      Text(next, style: t.bodyMedium?.copyWith(color: AppColors.textSoft)),
                      const SizedBox(height: 10),
                      OutlinedButton(
                        key: Key('wk-buy-${c.id}'),
                        onPressed: () => context.go('/joueur/personnages/${c.id}/xp'),
                        child: const Text('Préparer l’achat'),
                      ),
                    ],
                  ]),
                ),
              ],
          if (item.entry case final e?) ...[
            const SizedBox(height: 18),
            RuleDataView(cat: item.cat!, entry: e, focus: mine.firstOrNull, rb: rb),
          ],
          const SizedBox(height: 18),
          if (item.id == chronologyId)
            const ChronologyView()
          else if (body.trim().isEmpty)
            Text(item.isRule ? 'Le conte n’a pas encore rédigé cette page.' : 'Page vide.', style: t.bodyMedium?.copyWith(color: AppColors.textMuted))
          else
            WikiBody(body, headingKeys: keys, target: (title) => resolveLink(title, items)?.id, onOpen: (id) => context.go(nav.page(id))),
          if (page != null) ...[
            const SizedBox(height: 24),
            Text(pageFooter(page), style: t.bodySmall?.copyWith(color: AppColors.textMuted)),
          ],
        ]);
        final toc = heads.length < 2
            ? null
            : Panel(
                padding: const EdgeInsets.all(16),
                child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
                  const SectionTitle('Sur cette page'),
                  const SizedBox(height: 6),
                  for (final (n, h) in heads.indexed)
                    InkWell(
                      onTap: () {
                        final at = keys[n].currentContext;
                        if (at != null) Scrollable.ensureVisible(at, duration: const Duration(milliseconds: 300));
                      },
                      child: Padding(
                        padding: const EdgeInsets.symmetric(vertical: 6),
                        child: Text(h, style: t.bodyMedium?.copyWith(color: AppColors.link)),
                      ),
                    ),
                ]),
              );
        return PageBody(children: [
          if (manages && nav.staff && !nav.preview)
            Align(
              alignment: Alignment.centerRight,
              child: OutlinedButton(key: const Key('wk-edit'), onPressed: () => context.go('/conteur/wiki/edition/${item.id}'), child: const Text('Modifier')),
            ),
          if (toc != null && isWide(context))
            Row(crossAxisAlignment: CrossAxisAlignment.start, children: [
              SizedBox(width: 220, child: toc),
              const SizedBox(width: 32),
              Expanded(child: article),
            ])
          else
            article,
        ]);
      },
    );
  }
}
```
